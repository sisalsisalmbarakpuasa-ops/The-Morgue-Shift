local function placeGui(inst)
    local function try(p)
        if not p then return false end
        local ok = pcall(function() inst.Parent = p end)
        return ok and inst.Parent == p
    end
    if gethui then
        local ok, res = pcall(gethui)
        if ok and try(res) then return end
    end
    local ok, core = pcall(function() return game:GetService("CoreGui") end)
    if ok and try(core) then return end
    try(PlayerGui)
end

    for _, r in ipairs(roots) do
        for _, n in ipairs({ "MimiwMorgueGui", "MimiwMorgueESP" }) do
            local o = r and r:FindFirstChild(n)
            if o then o:Destroy() end
        end
    end
end)

    if old then pcall(old) end
end

-- ---------- DATA ----------
local candies = {}   -- [instance] = true  (nama berawalan "Candy")
local items = {}     -- [part] = true     (nama "TrickOrTreatItem")
local espObjs = {}   -- [instance] = {hl, bb}
local done = {}      -- [part] = waktu terakhir diambil

local function lname(o) return string.lower(o.Name) end

local function addESP(o)
    if espObjs[o] or not S.esp then return end
    local ok = pcall(function()
        local hl = Instance.new("Highlight")
        hl.Adornee = o
        hl.FillColor = RED
        hl.OutlineColor = RED
        hl.FillTransparency = 0.45
        hl.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        hl.Parent = ESPFolder
        local adorn = o:IsA("BasePart") and o or o:FindFirstChildWhichIsA("BasePart", true)
        local bb
        if adorn then
            bb = Instance.new("BillboardGui")
            bb.Adornee = adorn
            bb.AlwaysOnTop = true
            bb.Size = UDim2.new(0, 70, 0, 18)
            bb.StudsOffset = Vector3.new(0, 2, 0)
            local t = Instance.new("TextLabel", bb)
            t.Size = UDim2.new(1, 0, 1, 0)
            t.BackgroundTransparency = 1
            t.Text = "🍬 Candy"
            t.TextColor3 = RED
            t.TextStrokeTransparency = 0.3
            t.TextSize = 12
            t.Font = FONT
            bb.Parent = ESPFolder
        end
        espObjs[o] = { hl, bb }
    end)
    if not ok then espObjs[o] = nil end
end

local function clearESP(o)
    local e = espObjs[o]
    if e then
        for _, x in ipairs(e) do pcall(function() x:Destroy() end) end
        espObjs[o] = nil
    end
end

local function classify(o)
    if not (o:IsA("BasePart") or o:IsA("Model")) then return end
    local n = lname(o)
    if string.sub(n, 1, 5) == "candy" then
        candies[o] = true
        if S.esp then addESP(o) end
    end
    if o.Name == "TrickOrTreatItem" then
        items[o] = true
    end
end

local function unclassify(o)
    if candies[o] then candies[o] = nil clearESP(o) end
    if items[o] then items[o] = nil done[o] = nil end
end

conns[#conns + 1] = workspace.DescendantAdded:Connect(classify)
conns[#conns + 1] = workspace.DescendantRemoving:Connect(unclassify)
task.spawn(function()
    local n = 0
    for _, o in ipairs(workspace:GetDescendants()) do
        classify(o)
        n = n + 1
        if n % 800 == 0 then task.wait() end
    end
end)

local function setESP(on)
    S.esp = on
    if on then
        for o in pairs(candies) do addESP(o) end
    else
        for o in pairs(espObjs) do clearESP(o) end
    end
end

-- ---------- AUTO COLLECT ----------
local function myHrp()
    local c = LocalPlayer.Character
    return c and c:FindFirstChild("HumanoidRootPart")
end

local function promptOf(o)
    local p = o:FindFirstChildWhichIsA("ProximityPrompt", true)
    if not p and o.Parent then p = o.Parent:FindFirstChildWhichIsA("ProximityPrompt", true) end
    return p
end

local function posOf(o)
    if o:IsA("BasePart") then return o.Position end
    local ok, cf = pcall(function() return o:GetPivot() end)
    return ok and cf.Position or nil
end

local busy = false
local function collectNext()
    if busy then return false end
    local hrp = myHrp()
    if not hrp then return false end
    local now = os.clock()
    local best, bd, bp
    for o in pairs(items) do
        if o.Parent and (not done[o] or now - done[o] > 4) then
            local pos = posOf(o)
            local pp = promptOf(o)
            if pos and pp and pp.Enabled then
                local d = (pos - hrp.Position).Magnitude
                if not bd or d < bd then best, bd, bp = o, d, pp end
            end
        end
    end
    if not best then return false end
    busy = true
    done[best] = now
    pcall(function()
        local pos = posOf(best)
        hrp.CFrame = CFrame.new(pos + Vector3.new(0, 3, 0))
        hrp.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        task.wait(0.1)
        bp.HoldDuration = 0.1
        if fireproximityprompt then
            fireproximityprompt(bp)
        end
        task.wait(0.15)
    end)
    busy = false
    return true
end

task.spawn(function()
    while S.alive and sg.Parent do
        if S.auto then
            local ok = collectNext()
            task.wait(ok and 0.2 or 0.8)
        else
            task.wait(0.3)
        end
    end
end)

task.spawn(function()
    while S.alive and sg.Parent do
        local nc, ni = 0, 0
        for o in pairs(candies) do if o.Parent then nc = nc + 1 end end
        for o in pairs(items) do if o.Parent then ni = ni + 1 end end
        status.Text = "Candy: " .. nc .. "  |  TrickOrTreat: " .. ni
        task.wait(1)
    end
end)

    pcall(function() sg:Destroy() end)
end
